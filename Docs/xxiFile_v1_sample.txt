#!/bin/sh

. ./../../lib/common_func.sh
. ./../../lib/root_func.sh

# [appname] xxiFile
# xxiFile v1

check_depends() {
    check_command_exist make
    check_command_exist gcc
    check_command_exist wget
    check_command_exist tar
    check_command_exist nonexistent # Here is an error for testing
}

fetch_code() {
    mkdir -p /xxi/build/[appname] || fail "Failed to make build dir"
    cd /xxi/build/[appname] || fail "Failed to go to build dir when fetching code"
    wget [link] || fail "Failed to download code"
    tar xpvf [tarball] || fail "Failed to extract code"
}

build() {
    cd /xxi/build/[appname] || fail "Failed to go to build dir when building package"
    ./configure || fail "Failed to configure package"
    make || fail "Failed to make package"
    make install || fail "Failed to install package"
}
